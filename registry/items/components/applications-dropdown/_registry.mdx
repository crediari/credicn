Pass `systems` and `tools` as props. The search opens on demand, filters both lists locally, and
keeps the square app grid inside a styled `ScrollArea`. No SSO store required.

```tsx
import { ApplicationsDropdown } from "@/components/ui/applications-dropdown";

<ApplicationsDropdown
  user={{ name: "Ana Costa" }}
  systems={[{ id: "cards", name: "Cards", url: "https://cards.crediari.com.br" }]}
/>
```
